cmake_minimum_required(VERSION 3.22)
project(robot_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
foreach(name IN ITEMS zdt gripper)
    add_executable(test_${name} test_${name}.c ../App/Src/${name}.c)
    target_include_directories(test_${name} PRIVATE ../App/Inc)
    if(NOT MSVC)
        target_link_libraries(test_${name} PRIVATE m)
    endif()
    add_test(NAME ${name} COMMAND test_${name})
endforeach()

add_executable(test_runtime test_runtime.c ../App/Src/robot_runtime.c)
target_include_directories(test_runtime PRIVATE ../App/Inc ../Board/Inc)
add_test(NAME runtime COMMAND test_runtime)

foreach(name IN ITEMS zdt gripper runtime)
    if(MSVC)
        target_compile_options(test_${name} PRIVATE /W4 /WX /UNDEBUG)
    else()
        target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
endforeach()
